ជំពូកទី 01 • កិច្ចការអនុវត្តជាក់ស្តែង (Lab Practice)

ការអនុវត្ត៖ ការវិភាគគេហទំព័រអាជីពចំនួន ៣ (Analyze Three Professional Websites)

១. ព័ត៌មានទូទៅ និងគោលបំណង (Lab Overview & Objectives)

កិច្ចការអនុវត្តបន្ទប់ពិសោធន៍នៃជំពូកទី 01 នេះ ត្រូវបានរៀបចំឡើងដើម្បីផ្ដល់ឱ្យនិស្សិតនូវជំនាញជាក់ស្ដែងក្នុងការសរសេរកូដ ការវិភាគគេហទំព័រអាជីពចំនួន ៣ (Analyze Three Professional Websites) ដោយអនុលោមតាមបទដ្ឋានឧស្សាហកម្មវេបទំនើប W3C និង UI/UX Best Practices។

🎯 គោលបំណងចម្បងនៃការអនុវត្ត៖
  • ស្វែងយល់ និងអនុវត្តជាក់ស្ដែងលើរចនាសម្ព័ន្ធកូដ៖ Web Design, HTML, CSS, JavaScript, Client-Server
  • សរសេរកូដស្អាត (Clean Code) មានសណ្តាប់ធ្នាប់ និងឆ្លងកាត់ការផ្ទៀងផ្ទាត់ W3C Validator
  • ធានាភាពស៊ីសង្វាក់គ្នានៃការបង្ហាញលើ Mobile និង Desktop (Responsive Design)

function initTriadLab() {
  const toggleCss = document.getElementById('toggle-css');
  const toggleJs = document.getElementById('toggle-js');
  const sandboxView = document.getElementById('sandbox-view');
  const statusBadge = document.getElementById('sandbox-status-badge');
  const btnInteractive = document.getElementById('btn-interactive');
  const btnReset = document.getElementById('btn-reset-demo');
  const demoOutput = document.getElementById('demo-output');
  let clickCount = 0;

  function updateSandboxState() {
    // CSS state
    if (toggleCss.checked) {
      sandboxView.classList.remove('css-disabled');
      sandboxView.classList.add('css-enabled');
    } else {
      sandboxView.classList.remove('css-enabled');
      sandboxView.classList.add('css-disabled');
    }

    // JS state
    if (toggleJs.checked) {
      sandboxView.classList.add('js-enabled');
      sandboxView.classList.remove('js-disabled');
    } else {
      sandboxView.classList.remove('js-enabled');
      sandboxView.classList.add('js-disabled');
    }

    // Badge text
    if (toggleCss.checked && toggleJs.checked) {
      statusBadge.textContent = 'ស្ថានភាព: HTML + CSS + JS (គេហទំព័រពេញលេញ)';
      statusBadge.className = 'badge success';
    } else if (toggleCss.checked && !toggleJs.checked) {
      statusBadge.textContent = 'ស្ថានភាព: HTML + CSS (គេហទំព័រឋិតិវន្ត គ្មានអន្តរកម្ម)';
      statusBadge.className = 'badge warning';
    } else if (!toggleCss.checked && toggleJs.checked) {
      statusBadge.textContent = 'ស្ថានភាព: HTML + JS (មានអន្តរកម្ម ប៉ុន្តែគ្មានស្ទីល)';
      statusBadge.className = 'badge warning';
    } else {
      statusBadge.textContent = 'ស្ថានភាព: តែ HTML សុទ្ធ (គ្រោងឆ្អឹង Raw HTML)';
      statusBadge.className = 'badge';
    }
  }

  toggleCss.addEventListener('change', updateSandboxState);
  toggleJs.addEventListener('change', updateSandboxState);

  // Demo Interactive Button
  btnInteractive.addEventListener('click', () => {
    if (!toggleJs.checked) {
      alert('JavaScript ត្រូវបានបិទ! ប៊ូតុងមិនអាចឆ្លើយតបព្រឹត្តិការណ៍បានទេ។ សូមបើក JavaScript ឡើងវិញ។');
      return;
    }
    clickCount++;
    const now = new Date().toLocaleTimeString();
    demoOutput.innerHTML = `<strong>[JS Event Triggered]</strong> អ្នកបានចុចចំនួន <strong>${clickCount} ដង</strong>! ម៉ោងឆ្លើយតប: ${now}`;
    demoOutput.style.borderColor = '#10b981';
    demoOutput.style.backgroundColor = '#ecfdf5';
  });

  btnReset.addEventListener('click', () => {
    clickCount = 0;
    demoOutput.innerHTML = 'ចុចប៊ូតុងខាងលើដើម្បីមើលសកម្មភាព JavaScript...';
    demoOutput.style.borderColor = '#e2e8f0';
    demoOutput.style.backgroundColor = '#f8fafc';
  });
}

function initDevToolsSimulator() {
  const tabs = document.querySelectorAll('.dev-tab');
  const panels = document.querySelectorAll('.dev-panel');

  tabs.forEach(tab => {
    tab.addEventListener('click', () => {
      tabs.forEach(t => t.classList.remove('active'));
      panels.forEach(p => p.classList.remove('active'));

      tab.classList.add('active');
      const targetPanel = document.getElementById(`dev-panel-${tab.getAttribute('data-devtab')}`);
      if (targetPanel) targetPanel.classList.add('active');
    });
  });

  // Elements panel live controls
  const colorPicker = document.getElementById('dom-color-picker');
  const textInput = document.getElementById('dom-text-input');
  const targetHeading = document.getElementById('target-heading');

  if (colorPicker && targetHeading) {
    colorPicker.addEventListener('input', (e) => {
      targetHeading.style.color = e.target.value;
      addConsoleLog(`[Elements] Modified style.color to "${e.target.value}"`, 'info');
    });
  }

  if (textInput && targetHeading) {
    textInput.addEventListener('input', (e) => {
      targetHeading.textContent = e.target.value || 'ចំណងជើងសាកល្បង';
    });
  }

  // Console Panel Logic
  const consoleInput = document.getElementById('console-input');
  const btnRunConsole = document.getElementById('btn-run-console');
  const logsContainer = document.getElementById('console-logs');

  function addConsoleLog(message, type = 'log') {
    const entry = document.createElement('div');
    entry.className = `log-entry ${type}`;
    entry.textContent = message;
    logsContainer.appendChild(entry);
    logsContainer.scrollTop = logsContainer.scrollHeight;
  }

  function executeConsoleInput() {
    const val = consoleInput.value.trim();
    if (!val) return;

    addConsoleLog(`> ${val}`, 'log');

    try {
      // Safe evaluation for basic math/expressions
      const result = Function(`"use strict"; return (${val})`)();
      addConsoleLog(`<- ${result}`, 'info');
    } catch (err) {
      addConsoleLog(`Uncaught ${err.name}: ${err.message}`, 'error');
    }

    consoleInput.value = '';
  }

  if (btnRunConsole) btnRunConsole.addEventListener('click', executeConsoleInput);
  if (consoleInput) {
    consoleInput.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') executeConsoleInput();
    });
  }

  // Network Panel Simulated Fetch
  const btnFetchMock = document.getElementById('btn-fetch-mock');
  const btnClearNetwork = document.getElementById('btn-clear-network');
  const tableBody = document.getElementById('network-table-body');
  const counterEl = document.getElementById('network-counter');
  let mockCount = 3;

  if (btnFetchMock) {
    btnFetchMock.addEventListener('click', () => {
      mockCount++;
      const latency = Math.floor(Math.random() * 45) + 15;
      const size = (Math.random() * 4 + 1.2).toFixed(1);

      const row = document.createElement('tr');
      row.innerHTML = `
        <td>/api/v1/courses/chapter-0${mockCount}.json</td>
        <td>GET</td>
        <td><span class="badge success">200 OK</span></td>
        <td>fetch / json</td>
        <td>${size} KB</td>
        <td>${latency} ms</td>
      `;
      tableBody.appendChild(row);
      counterEl.textContent = `${mockCount} Requests | ${(mockCount * 3.8).toFixed(1)} KB transferred`;
      addConsoleLog(`[Network] Simulated HTTP GET /api/v1/courses/chapter-0${mockCount}.json 200 OK (${latency}ms)`, 'info');
    });
  }

  if (btnClearNetwork) {
    btnClearNetwork.addEventListener('click', () => {
      tableBody.innerHTML = '';
      mockCount = 0;
      counterEl.textContent = '0 Requests | 0 KB transferred';
      addConsoleLog('[Network] Network log cleared.', 'warn');
    });
  }
}

function initTriadLab() {
  const toggleCss = document.getElementById('toggle-css');
  const toggleJs = document.getElementById('toggle-js');
  const sandboxView = document.getElementById('sandbox-view');
  const statusBadge = document.getElementById('sandbox-status-badge');
  const btnInteractive = document.getElementById('btn-interactive');
  const btnReset = document.getElementById('btn-reset-demo');
  const demoOutput = document.getElementById('demo-output');
  let clickCount = 0;

  function updateSandboxState() {
    // CSS state
    if (toggleCss.checked) {
      sandboxView.classList.remove('css-disabled');
      sandboxView.classList.add('css-enabled');
    } else {
      sandboxView.classList.remove('css-enabled');
      sandboxView.classList.add('css-disabled');
    }

    // JS state
    if (toggleJs.checked) {
      sandboxView.classList.add('js-enabled');
      sandboxView.classList.remove('js-disabled');
    } else {
      sandboxView.classList.remove('js-enabled');
      sandboxView.classList.add('js-disabled');
    }

    // Badge text
    if (toggleCss.checked && toggleJs.checked) {
      statusBadge.textContent = 'ស្ថានភាព: HTML + CSS + JS (គេហទំព័រពេញលេញ)';
      statusBadge.className = 'badge success';
    } else if (toggleCss.checked && !toggleJs.checked) {
      statusBadge.textContent = 'ស្ថានភាព: HTML + CSS (គេហទំព័រឋិតិវន្ត គ្មានអន្តរកម្ម)';
      statusBadge.className = 'badge warning';
    } else if (!toggleCss.checked && toggleJs.checked) {
      statusBadge.textContent = 'ស្ថានភាព: HTML + JS (មានអន្តរកម្ម ប៉ុន្តែគ្មានស្ទីល)';
      statusBadge.className = 'badge warning';
    } else {
      statusBadge.textContent = 'ស្ថានភាព: តែ HTML សុទ្ធ (គ្រោងឆ្អឹង Raw HTML)';
      statusBadge.className = 'badge';
    }
  }

  toggleCss.addEventListener('change', updateSandboxState);
  toggleJs.addEventListener('change', updateSandboxState);

  // Demo Interactive Button
  btnInteractive.addEventListener('click', () => {
    if (!toggleJs.checked) {
      alert('JavaScript ត្រូវបានបិទ! ប៊ូតុងមិនអាចឆ្លើយតបព្រឹត្តិការណ៍បានទេ។ សូមបើក JavaScript ឡើងវិញ។');
      return;
    }
    clickCount++;
    const now = new Date().toLocaleTimeString();
    demoOutput.innerHTML = `<strong>[JS Event Triggered]</strong> អ្នកបានចុចចំនួន <strong>${clickCount} ដង</strong>! ម៉ោងឆ្លើយតប: ${now}`;
    demoOutput.style.borderColor = '#10b981';
    demoOutput.style.backgroundColor = '#ecfdf5';
  });

  btnReset.addEventListener('click', () => {
    clickCount = 0;
    demoOutput.innerHTML = 'ចុចប៊ូតុងខាងលើដើម្បីមើលសកម្មភាព JavaScript...';
    demoOutput.style.borderColor = '#e2e8f0';
    demoOutput.style.backgroundColor = '#f8fafc';
  });
}

      }
    });
  });

function initDevToolsSimulator() {
  const tabs = document.querySelectorAll('.dev-tab');
  const panels = document.querySelectorAll('.dev-panel');

  tabs.forEach(tab => {
    tab.addEventListener('click', () => {
      tabs.forEach(t => t.classList.remove('active'));
      panels.forEach(p => p.classList.remove('active'));

      tab.classList.add('active');
      const targetPanel = document.getElementById(`dev-panel-${tab.getAttribute('data-devtab')}`);
      if (targetPanel) targetPanel.classList.add('active');
    });
  });

  // Elements panel live controls
  const colorPicker = document.getElementById('dom-color-picker');
  const textInput = document.getElementById('dom-text-input');
  const targetHeading = document.getElementById('target-heading');

  if (colorPicker && targetHeading) {
    colorPicker.addEventListener('input', (e) => {
      targetHeading.style.color = e.target.value;
      addConsoleLog(`[Elements] Modified style.color to "${e.target.value}"`, 'info');
    });
  }

  if (textInput && targetHeading) {
    textInput.addEventListener('input', (e) => {
      targetHeading.textContent = e.target.value || 'ចំណងជើងសាកល្បង';
    });
  }

  // Console Panel Logic
  const consoleInput = document.getElementById('console-input');
  const btnRunConsole = document.getElementById('btn-run-console');
  const logsContainer = document.getElementById('console-logs');

  function addConsoleLog(message, type = 'log') {
    const entry = document.createElement('div');
    entry.className = `log-entry ${type}`;
    entry.textContent = message;
    logsContainer.appendChild(entry);
    logsContainer.scrollTop = logsContainer.scrollHeight;
  }

  function executeConsoleInput() {
    const val = consoleInput.value.trim();
    if (!val) return;

    addConsoleLog(`> ${val}`, 'log');

    try {
      // Safe evaluation for basic math/expressions
      const result = Function(`"use strict"; return (${val})`)();
      addConsoleLog(`<- ${result}`, 'info');
    } catch (err) {
      addConsoleLog(`Uncaught ${err.name}: ${err.message}`, 'error');
    }

    consoleInput.value = '';
  }

  if (btnRunConsole) btnRunConsole.addEventListener('click', executeConsoleInput);
  if (consoleInput) {
    consoleInput.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') executeConsoleInput();
    });
  }

  // Network Panel Simulated Fetch
  const btnFetchMock = document.getElementById('btn-fetch-mock');
  const btnClearNetwork = document.getElementById('btn-clear-network');
  const tableBody = document.getElementById('network-table-body');
  const counterEl = document.getElementById('network-counter');
  let mockCount = 3;

  if (btnFetchMock) {
    btnFetchMock.addEventListener('click', () => {
      mockCount++;
      const latency = Math.floor(Math.random() * 45) + 15;
      const size = (Math.random() * 4 + 1.2).toFixed(1);

      const row = document.createElement('tr');
      row.innerHTML = `
        <td>/api/v1/courses/chapter-0${mockCount}.json</td>
        <td>GET</td>
        <td><span class="badge success">200 OK</span></td>
        <td>fetch / json</td>
        <td>${size} KB</td>
        <td>${latency} ms</td>
      `;
      tableBody.appendChild(row);
      counterEl.textContent = `${mockCount} Requests | ${(mockCount * 3.8).toFixed(1)} KB transferred`;
      addConsoleLog(`[Network] Simulated HTTP GET /api/v1/courses/chapter-0${mockCount}.json 200 OK (${latency}ms)`, 'info');
    });
  }

  if (btnClearNetwork) {
    btnClearNetwork.addEventListener('click', () => {
      tableBody.innerHTML = '';
      mockCount = 0;
      counterEl.textContent = '0 Requests | 0 KB transferred';
      addConsoleLog('[Network] Network log cleared.', 'warn');
    });
  }
}

១. រចនាសម្ព័ន្ធកូដ & HTML5 Semantics ២៥ ពិន្ទុ ប្រើប្រាស់ Semantic Elements ត្រឹមត្រូវតាមស្ដង់ដារ W3C, DOCTYPE, ឋានានុក្រម Headings (H1-H6) ច្បាស់លាស់ និងរៀបចំ Indentation ស្អាតបាត។
២. រចនាប័ទ្ម CSS & Responsive Design ២៥ ពិន្ទុ ប្រើប្រាស់ CSS Variables (:root), Box Model, Flexbox/Grid បានត្រឹមត្រូវ និងមាន Media Queries គាំទ្រការបង្ហាញលើ Mobile បានយ៉ាងរលូន។
៣. ភាពត្រឹមត្រូវនៃមុខងារអនុវត្ត ២៥ ពិន្ទុ សមាសភាគស្នូលនៃមេរៀន (ការវិភាគគេហទំព័រអាជីពចំនួន ៣) ដំណើរការពេញលេញ គ្មាន Error និងឆ្លើយតបទៅនឹងគោលបំណងនៃមន្ទីរពិសោធន៍។
៤. សោភ័ណភាព UI/UX & ភាពងាយស្រួល ១៥ ពិន្ទុ ការជ្រើសរើសកម្រិតពណ៌ផ្ទុយត្រឹមត្រូវតាម WCAG, ការរៀបចំគម្លាត Spacing ស្មើដៃ និងទំហំអក្សរងាយស្រួលអានលើគ្រប់ឧបករណ៍។
