១. ព័ត៌មានទូទៅ និងគោលបំណង (Lab Overview & Objectives)

កិច្ចការអនុវត្តបន្ទប់ពិសោធន៍នៃជំពូកទី 03 នេះ ត្រូវបានរៀបចំឡើងដើម្បីផ្ដល់ឱ្យនិស្សិតនូវជំនាញជាក់ស្ដែងក្នុងការសរសេរកូដ ការបង្កើតលំហូរអ្នកប្រើប្រាស់សម្រាប់គេហទំព័រ E-Commerce (Create a User Flow for an E-Commerce Website) ដោយអនុលោមតាមបទដ្ឋានឧស្សាហកម្មវេបទំនើប W3C និង UI/UX Best Practices។

🎯 គោលបំណងចម្បងនៃការអនុវត្ត៖
  • ស្វែងយល់ និងអនុវត្តជាក់ស្ដែងលើរចនាសម្ព័ន្ធកូដ៖ UI, UX, User Flow, User Journey, Usability
  • សរសេរកូដស្អាត (Clean Code) មានសណ្តាប់ធ្នាប់ និងឆ្លងកាត់ការផ្ទៀងផ្ទាត់ W3C Validator
  • ធានាភាពស៊ីសង្វាក់គ្នានៃការបង្ហាញលើ Mobile និង Desktop (Responsive Design)

   ============================================================ */
function initUserFlowController() {
  const stepItems = document.querySelectorAll('.step-item');
  const screens = document.querySelectorAll('.flow-screen');

  window.goToStep = function(stepNum) {
    if (stepNum < 1 || stepNum > 5) return;
    appState.currentStep = stepNum;

    // Update screens
    screens.forEach(s => s.classList.remove('active'));
    const targetScreen = document.getElementById(`screen-step-${stepNum}`);
    if (targetScreen) targetScreen.classList.add('active');

    // Update progress bar
    stepItems.forEach(item => {
      const itemStep = parseInt(item.getAttribute('data-step'), 10);
      item.classList.remove('active', 'completed');
      if (itemStep === stepNum) {
        item.classList.add('active');
      } else if (itemStep < stepNum) {
        item.classList.add('completed');
      }
    });

    // Scroll to top of main
    window.scrollTo({ top: 120, behavior: 'smooth' });

    // Toast Feedback
    showToast(`កំពុងស្ថិតនៅជំហានទី ${stepNum}៖ ${getStepTitle(stepNum)}`);
  };

  function getStepTitle(num) {
    const titles = {
      1: 'ជ្រើសរើសផលិតផល',
      2: 'ពិនិត្យកន្ត្រកទំនិញ',
      3: 'អាសយដ្ឋានដឹកជញ្ជូន',
      4: 'ការទូទាត់ប្រាក់ (Payment)',
      5: 'ការបញ្ជាទិញជោគជ័យ'
    };
    return titles[num] || '';
  }

  // Handle Back Buttons
  document.querySelectorAll('.btn-back').forEach(btn => {
    btn.addEventListener('click', () => {
      const toStep = parseInt(btn.getAttribute('data-to'), 10);
      goToStep(toStep);
    });
  });

  // Step 1 Add to Cart -> Step 2
  const btnAddToCart = document.getElementById('btn-add-to-cart');
  if (btnAddToCart) {
    btnAddToCart.addEventListener('click', () => {
      updateCartSummaryUI();
      goToStep(2);
    });
  }

  // Step 2 Proceed to Shipping -> Step 3
  const btnProceedShipping = document.getElementById('btn-proceed-to-shipping');
  if (btnProceedShipping) {
    btnProceedShipping.addEventListener('click', () => {
      goToStep(3);
    });
  }

  // Step 5 Restart Flow
  const btnRestart = document.getElementById('btn-restart-flow');
  if (btnRestart) {
    btnRestart.addEventListener('click', () => {
      goToStep(1);
    });
  }
}

function initProductConfigurator() {
  const thumbBtns = document.querySelectorAll('.thumb-btn');
  const colorBadge = document.getElementById('current-color-badge');
  const mainIcon = document.querySelector('.product-icon-large');
  const storageRadios = document.querySelectorAll('input[name="storage"]');
  const btnMinus = document.getElementById('btn-qty-minus');
  const btnPlus = document.getElementById('btn-qty-plus');
  const inputQty = document.getElementById('input-qty');
  const priceDisplay = document.querySelector('.price-current');

  function updatePriceDisplay() {
    const unitPrice = appState.basePrice + appState.storageExtra;
    priceDisplay.textContent = `$${unitPrice.toFixed(2)}`;
  }

  // Color Switcher
  thumbBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      thumbBtns.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');

      const color = btn.getAttribute('data-color');
      const img = btn.getAttribute('data-img');
      appState.selectedColor = color;
      colorBadge.textContent = `ពណ៌៖ ${color}`;
      mainIcon.textContent = img;
      showToast(`បានជ្រើសរើសពណ៌៖ ${color}`);
    });
  });

  // Storage selection
  storageRadios.forEach(radio => {
    radio.addEventListener('change', () => {
      if (radio.value === '1TB') {
        appState.storageExtra = 150;
        appState.selectedStorage = '1TB';
      } else {
        appState.storageExtra = 0;
        appState.selectedStorage = '512GB';
      }
      updatePriceDisplay();
      showToast(`បានជ្រើសរើសទំហំផ្ទុក៖ ${appState.selectedStorage}`);
    });
  });

  // Quantity stepper
  if (btnMinus && btnPlus && inputQty) {
    btnMinus.addEventListener('click', () => {
      if (appState.quantity > 1) {
        appState.quantity--;
        inputQty.value = appState.quantity;
      }
    });

    btnPlus.addEventListener('click', () => {
      if (appState.quantity < 5) {
        appState.quantity++;
        inputQty.value = appState.quantity;
      } else {
        showToast('បរិមាណអតិបរមាគឺ ៥ គ្រឿងសម្រាប់ម្នាក់!');
      }
    });
  }
}

function initFormValidation() {
  const form = document.getElementById('shipping-form');
  const inputName = document.getElementById('cust-name');
  const inputPhone = document.getElementById('cust-phone');
  const inputEmail = document.getElementById('cust-email');
  const inputCity = document.getElementById('cust-city');
  const inputAddress = document.getElementById('cust-address');

  const errName = document.getElementById('err-name');
  const errPhone = document.getElementById('err-phone');
  const errEmail = document.getElementById('err-email');
  const errAddress = document.getElementById('err-address');

  function validateField(input, errElement, condition, errorMsg) {
    if (!condition) {
      input.classList.add('invalid');
      errElement.textContent = errorMsg;
      return false;
    } else {
      input.classList.remove('invalid');
      errElement.textContent = '';
      return true;
    }
  }

  // Real-time Inline Validations
  inputName.addEventListener('input', () => {
    validateField(inputName, errName, inputName.value.trim().length >= 3, 'ឈ្មោះត្រូវមានយ៉ាងតិច ៣ តួអក្សរ!');
  });

  inputPhone.addEventListener('input', () => {
    const phoneRegex = /^[0-9]{8,11}$/;
    validateField(inputPhone, errPhone, phoneRegex.test(inputPhone.value.trim()), 'លេខទូរស័ព្ទត្រូវតែជាលេខ ៨ ដល់ ១១ ខ្ទង់!');
  });

  inputEmail.addEventListener('input', () => {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    validateField(inputEmail, errEmail, emailRegex.test(inputEmail.value.trim()), 'ទម្រង់អ៊ីមែលមិនត្រឹមត្រូវ (ឧ. user@domain.com)!');
  });

  inputAddress.addEventListener('input', () => {
    validateField(inputAddress, errAddress, inputAddress.value.trim().length >= 5, 'សូមបំពេញអាសយដ្ឋានឱ្យបានលម្អិត!');
  });

  // Form Submit
  if (form) {
    form.addEventListener('submit', (e) => {
      e.preventDefault();

      const isNameValid = validateField(inputName, errName, inputName.value.trim().length >= 3, 'សូមបញ្ចូលឈ្មោះពេញរបស់អ្នក!');
      const isPhoneValid = validateField(inputPhone, errPhone, /^[0-9]{8,11}$/.test(inputPhone.value.trim()), 'សូមបញ្ចូលលេខទូរស័ព្ទត្រឹមត្រូវ!');
      const isEmailValid = validateField(inputEmail, errEmail, /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(inputEmail.value.trim()), 'សូមបញ្ចូលអ៊ីមែលត្រឹមត្រូវ!');
      const isAddressValid = validateField(inputAddress, errAddress, inputAddress.value.trim().length >= 5, 'សូមបញ្ចូលអាសយដ្ឋានដឹកជញ្ជូន!');

      if (isNameValid && isPhoneValid && isEmailValid && isAddressValid) {
        // Save to state
        appState.customer.name = inputName.value.trim();
        appState.customer.phone = inputPhone.value.trim();
        appState.customer.email = inputEmail.value.trim();
        appState.customer.city = inputCity.value;
        appState.customer.address = inputAddress.value.trim();

        // Update step 4 UI
        document.getElementById('final-cust-name').textContent = appState.customer.name;
        document.getElementById('final-cust-phone').textContent = appState.customer.phone;
        document.getElementById('final-cust-city').textContent = `${appState.customer.address}, ${appState.customer.city}`;

        showToast('ព័ត៌មានដឹកជញ្ជូនត្រឹមត្រូវ! កំពុងបន្តទៅទូទាត់ប្រាក់...');
        goToStep(4);
      } else {
        showToast('⚠️ សូមពិនិត្យមើលកំហុសក្នុងទម្រង់បែបបទ (Error Prevention)!');
      }
    });
  }
}

function initPaymentAndCountdown() {
  const payOptions = document.querySelectorAll('.payment-option');
  const btnConfirmPayment = document.getElementById('btn-confirm-payment');
  const timerDisplay = document.getElementById('qr-timer');
  let countdownSeconds = 300; // 5 minutes

  // Toggle payment method selection
  payOptions.forEach(opt => {
    opt.addEventListener('click', () => {
      payOptions.forEach(o => o.classList.remove('selected'));
      opt.classList.add('selected');
      const radio = opt.querySelector('input[type="radio"]');
      if (radio) radio.checked = true;

      const method = opt.getAttribute('data-method');
      if (method === 'khqr') {
        document.getElementById('khqr-box').style.display = 'block';
        showToast('បានជ្រើសរើសវិធីសាស្ត្រ Bakong KHQR');
      } else {
        document.getElementById('khqr-box').style.display = 'none';
        showToast('បានជ្រើសរើសវិធីសាស្ត្រ Visa / MasterCard');
      }
    });
  });

  // Countdown simulation
  setInterval(() => {
    if (countdownSeconds > 0) {
      countdownSeconds--;
      const mins = Math.floor(countdownSeconds / 60);
      const secs = countdownSeconds % 60;
      if (timerDisplay) {
        timerDisplay.textContent = `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
      }
    }
  }, 1000);

  // Confirm payment
  if (btnConfirmPayment) {
    btnConfirmPayment.addEventListener('click', () => {
      btnConfirmPayment.innerHTML = '⏳ កំពុងដំណើរការទូទាត់...';
      btnConfirmPayment.disabled = true;

      setTimeout(() => {
        btnConfirmPayment.innerHTML = '🔒 បញ្ជាក់ការទូទាត់ និងបញ្ចប់ការបញ្ជាទិញ';
        btnConfirmPayment.disabled = false;

        // Generate receipt values
        document.getElementById('order-id').textContent = `#ORD-2026-${Math.floor(1000 + Math.random() * 9000)}`;
        document.getElementById('receipt-date').textContent = new Date().toLocaleDateString('km-KH', { dateStyle: 'long' });
        document.getElementById('receipt-item').textContent = `TechPro Laptop (${appState.selectedColor}, ${appState.selectedStorage}) x ${appState.quantity}`;

        goToStep(5);
        showToast('🎉 ការទូទាត់ជោគជ័យ! បញ្ជាទិញបានសម្រេច។');
      }, 1200);
    });
  }
}

function initModalsAndHUD() {
  const toast = document.getElementById('toast-notification');
  const toastText = document.getElementById('toast-text');

  window.showToast = function(msg) {
    if (!toast || !toastText) return;
    toastText.textContent = msg;
    toast.classList.add('show');
    setTimeout(() => {
      toast.classList.remove('show');
    }, 3200);
  };

  // Flow Diagram Modal
  const flowModal = document.getElementById('flow-modal');
  const btnOpenFlow1 = document.getElementById('btn-view-userflow-diagram');
  const btnOpenFlow2 = document.getElementById('btn-show-flow-modal-2');
  const btnCloseFlow = document.getElementById('btn-close-flow-modal');
  const btnCloseFlowFooter = document.getElementById('btn-close-flow-footer');

  function openFlowModal() {
    if (flowModal) flowModal.classList.add('open');
  }

  function closeFlowModal() {
    if (flowModal) flowModal.classList.remove('open');
  }

  if (btnOpenFlow1) btnOpenFlow1.addEventListener('click', openFlowModal);
  if (btnOpenFlow2) btnOpenFlow2.addEventListener('click', openFlowModal);
  if (btnCloseFlow) btnCloseFlow.addEventListener('click', closeFlowModal);
  if (btnCloseFlowFooter) btnCloseFlowFooter.addEventListener('click', closeFlowModal);

  window.addEventListener('click', (e) => {
    if (e.target === flowModal) closeFlowModal();
  });

  // Toggle HUD minimize
  const btnToggleHud = document.getElementById('btn-toggle-hud');
  const hudContent = document.getElementById('hud-content');
  let hudMinimized = false;

  if (btnToggleHud && hudContent) {
    btnToggleHud.addEventListener('click', () => {
      hudMinimized = !hudMinimized;
      if (hudMinimized) {
        hudContent.style.display = 'none';
        btnToggleHud.textContent = '+';
      } else {
        hudContent.style.display = 'block';
        btnToggleHud.textContent = '_';
      }
    });
  }
}

function initUserFlowController() {
  const stepItems = document.querySelectorAll('.step-item');
  const screens = document.querySelectorAll('.flow-screen');

  window.goToStep = function(stepNum) {
    if (stepNum < 1 || stepNum > 5) return;
    appState.currentStep = stepNum;

    // Update screens
    screens.forEach(s => s.classList.remove('active'));
    const targetScreen = document.getElementById(`screen-step-${stepNum}`);
    if (targetScreen) targetScreen.classList.add('active');

    // Update progress bar
    stepItems.forEach(item => {
      const itemStep = parseInt(item.getAttribute('data-step'), 10);
      item.classList.remove('active', 'completed');
      if (itemStep === stepNum) {
        item.classList.add('active');
      } else if (itemStep < stepNum) {
        item.classList.add('completed');
      }
    });

    // Scroll to top of main
    window.scrollTo({ top: 120, behavior: 'smooth' });

    // Toast Feedback
    showToast(`កំពុងស្ថិតនៅជំហានទី ${stepNum}៖ ${getStepTitle(stepNum)}`);
  };

  function getStepTitle(num) {
    const titles = {
      1: 'ជ្រើសរើសផលិតផល',
      2: 'ពិនិត្យកន្ត្រកទំនិញ',
      3: 'អាសយដ្ឋានដឹកជញ្ជូន',
      4: 'ការទូទាត់ប្រាក់ (Payment)',
      5: 'ការបញ្ជាទិញជោគជ័យ'
    };
    return titles[num] || '';
  }

  // Handle Back Buttons
  document.querySelectorAll('.btn-back').forEach(btn => {
    btn.addEventListener('click', () => {
      const toStep = parseInt(btn.getAttribute('data-to'), 10);
      goToStep(toStep);
    });
  });

  // Step 1 Add to Cart -> Step 2
  const btnAddToCart = document.getElementById('btn-add-to-cart');
  if (btnAddToCart) {
    btnAddToCart.addEventListener('click', () => {
      updateCartSummaryUI();
      goToStep(2);
    });
  }

  // Step 2 Proceed to Shipping -> Step 3
  const btnProceedShipping = document.getElementById('btn-proceed-to-shipping');
  if (btnProceedShipping) {
    btnProceedShipping.addEventListener('click', () => {
      goToStep(3);
    });
  }

  // Step 5 Restart Flow
  const btnRestart = document.getElementById('btn-restart-flow');
  if (btnRestart) {
    btnRestart.addEventListener('click', () => {
      goToStep(1);
    });
  }
}

function initProductConfigurator() {
  const thumbBtns = document.querySelectorAll('.thumb-btn');
  const colorBadge = document.getElementById('current-color-badge');
  const mainIcon = document.querySelector('.product-icon-large');
  const storageRadios = document.querySelectorAll('input[name="storage"]');
  const btnMinus = document.getElementById('btn-qty-minus');
  const btnPlus = document.getElementById('btn-qty-plus');
  const inputQty = document.getElementById('input-qty');
  const priceDisplay = document.querySelector('.price-current');

  function updatePriceDisplay() {
    const unitPrice = appState.basePrice + appState.storageExtra;
    priceDisplay.textContent = `$${unitPrice.toFixed(2)}`;
  }

  // Color Switcher
  thumbBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      thumbBtns.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');

      const color = btn.getAttribute('data-color');
      const img = btn.getAttribute('data-img');
      appState.selectedColor = color;
      colorBadge.textContent = `ពណ៌៖ ${color}`;
      mainIcon.textContent = img;
      showToast(`បានជ្រើសរើសពណ៌៖ ${color}`);
    });
  });

  // Storage selection
  storageRadios.forEach(radio => {
    radio.addEventListener('change', () => {
      if (radio.value === '1TB') {
        appState.storageExtra = 150;
        appState.selectedStorage = '1TB';
      } else {
        appState.storageExtra = 0;
        appState.selectedStorage = '512GB';
      }
      updatePriceDisplay();
      showToast(`បានជ្រើសរើសទំហំផ្ទុក៖ ${appState.selectedStorage}`);
    });
  });

  // Quantity stepper
  if (btnMinus && btnPlus && inputQty) {
    btnMinus.addEventListener('click', () => {
      if (appState.quantity > 1) {
        appState.quantity--;
        inputQty.value = appState.quantity;
      }
    });

    btnPlus.addEventListener('click', () => {
      if (appState.quantity < 5) {
        appState.quantity++;
        inputQty.value = appState.quantity;
      } else {
        showToast('បរិមាណអតិបរមាគឺ ៥ គ្រឿងសម្រាប់ម្នាក់!');
      }
    });
  }
}

function initCartAndCoupon() {

  const couponFeedback = document.getElementById('coupon-feedback');
  const btnClearCart = document.getElementById('btn-clear-cart');

    const unitPrice = appState.basePrice + appState.storageExtra;
    const subtotal = unitPrice * appState.quantity;
    const discountAmount = subtotal * appState.discountPercent;
    const total = subtotal - discountAmount;

    // Update Cart Screen UI
    document.getElementById('cart-item-specs').textContent = `ពណ៌៖ ${appState.selectedColor} | ${appState.selectedStorage} SSD`;
    document.getElementById('cart-item-unit-price').textContent = `$${unitPrice.toFixed(2)}`;
    document.getElementById('cart-qty-display').textContent = appState.quantity;
    document.getElementById('cart-item-subtotal').textContent = `$${subtotal.toFixed(2)}`;

    document.getElementById('sum-items').textContent = `$${subtotal.toFixed(2)}`;
    document.getElementById('sum-discount').textContent = `-$${discountAmount.toFixed(2)}`;
    document.getElementById('sum-total').textContent = `$${total.toFixed(2)}`;

    // Update Payment Screen UI
    document.getElementById('khqr-amount-display').textContent = `$${total.toFixed(2)}`;
    document.getElementById('final-total').textContent = `$${total.toFixed(2)}`;
    document.getElementById('receipt-amount').textContent = `$${total.toFixed(2)}`;
  };

  if (btnApplyCoupon && couponInput && couponFeedback) {
    btnApplyCoupon.addEventListener('click', () => {
      const code = couponInput.value.trim().toUpperCase();
      if (code === 'STUDENT2026') {
        appState.discountPercent = 0.10; // 10%
        couponFeedback.textContent = '✅ លេខកូដត្រឹមត្រូវ! អ្នកទទួលបានការបញ្ចុះតម្លៃ ១០%។';

        showToast('ទទួលបានការបញ្ចុះតម្លៃ ១០% ដោយជោគជ័យ!');
      } else if (code === '') {
        couponFeedback.textContent = '⚠️ សូមវាយបញ្ចូលលេខកូដ!';
        couponFeedback.style.color = '#ef4444';
      } else {
        couponFeedback.textContent = '❌ លេខកូដមិនត្រឹមត្រូវ ឬផុតកំណត់!';
        couponFeedback.style.color = '#ef4444';
      }
    });

  if (btnClearCart) {
    btnClearCart.addEventListener('click', () => {

        goToStep(1);
      }
    });
  }
}

function initFormValidation() {
  const form = document.getElementById('shipping-form');
  const inputName = document.getElementById('cust-name');
  const inputPhone = document.getElementById('cust-phone');
  const inputEmail = document.getElementById('cust-email');
  const inputCity = document.getElementById('cust-city');
  const inputAddress = document.getElementById('cust-address');

  const errName = document.getElementById('err-name');
  const errPhone = document.getElementById('err-phone');
  const errEmail = document.getElementById('err-email');
  const errAddress = document.getElementById('err-address');

  function validateField(input, errElement, condition, errorMsg) {
    if (!condition) {
      input.classList.add('invalid');
      errElement.textContent = errorMsg;
      return false;
    } else {
      input.classList.remove('invalid');
      errElement.textContent = '';
      return true;
    }
  }

  // Real-time Inline Validations
  inputName.addEventListener('input', () => {
    validateField(inputName, errName, inputName.value.trim().length >= 3, 'ឈ្មោះត្រូវមានយ៉ាងតិច ៣ តួអក្សរ!');
  });

  inputPhone.addEventListener('input', () => {
    const phoneRegex = /^[0-9]{8,11}$/;
    validateField(inputPhone, errPhone, phoneRegex.test(inputPhone.value.trim()), 'លេខទូរស័ព្ទត្រូវតែជាលេខ ៨ ដល់ ១១ ខ្ទង់!');
  });

  inputEmail.addEventListener('input', () => {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    validateField(inputEmail, errEmail, emailRegex.test(inputEmail.value.trim()), 'ទម្រង់អ៊ីមែលមិនត្រឹមត្រូវ (ឧ. user@domain.com)!');
  });

  inputAddress.addEventListener('input', () => {
    validateField(inputAddress, errAddress, inputAddress.value.trim().length >= 5, 'សូមបំពេញអាសយដ្ឋានឱ្យបានលម្អិត!');
  });

  // Form Submit
  if (form) {
    form.addEventListener('submit', (e) => {
      e.preventDefault();

      const isNameValid = validateField(inputName, errName, inputName.value.trim().length >= 3, 'សូមបញ្ចូលឈ្មោះពេញរបស់អ្នក!');
      const isPhoneValid = validateField(inputPhone, errPhone, /^[0-9]{8,11}$/.test(inputPhone.value.trim()), 'សូមបញ្ចូលលេខទូរស័ព្ទត្រឹមត្រូវ!');
      const isEmailValid = validateField(inputEmail, errEmail, /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(inputEmail.value.trim()), 'សូមបញ្ចូលអ៊ីមែលត្រឹមត្រូវ!');
      const isAddressValid = validateField(inputAddress, errAddress, inputAddress.value.trim().length >= 5, 'សូមបញ្ចូលអាសយដ្ឋានដឹកជញ្ជូន!');

      if (isNameValid && isPhoneValid && isEmailValid && isAddressValid) {
        // Save to state
        appState.customer.name = inputName.value.trim();
        appState.customer.phone = inputPhone.value.trim();
        appState.customer.email = inputEmail.value.trim();
        appState.customer.city = inputCity.value;
        appState.customer.address = inputAddress.value.trim();

        // Update step 4 UI
        document.getElementById('final-cust-name').textContent = appState.customer.name;
        document.getElementById('final-cust-phone').textContent = appState.customer.phone;
        document.getElementById('final-cust-city').textContent = `${appState.customer.address}, ${appState.customer.city}`;

        showToast('ព័ត៌មានដឹកជញ្ជូនត្រឹមត្រូវ! កំពុងបន្តទៅទូទាត់ប្រាក់...');
        goToStep(4);
      } else {
        showToast('⚠️ សូមពិនិត្យមើលកំហុសក្នុងទម្រង់បែបបទ (Error Prevention)!');
      }
    });
  }
}

function initPaymentAndCountdown() {
  const payOptions = document.querySelectorAll('.payment-option');
  const btnConfirmPayment = document.getElementById('btn-confirm-payment');
  const timerDisplay = document.getElementById('qr-timer');
  let countdownSeconds = 300; // 5 minutes

  // Toggle payment method selection
  payOptions.forEach(opt => {
    opt.addEventListener('click', () => {
      payOptions.forEach(o => o.classList.remove('selected'));
      opt.classList.add('selected');
      const radio = opt.querySelector('input[type="radio"]');
      if (radio) radio.checked = true;

      const method = opt.getAttribute('data-method');
      if (method === 'khqr') {
        document.getElementById('khqr-box').style.display = 'block';
        showToast('បានជ្រើសរើសវិធីសាស្ត្រ Bakong KHQR');
      } else {
        document.getElementById('khqr-box').style.display = 'none';
        showToast('បានជ្រើសរើសវិធីសាស្ត្រ Visa / MasterCard');
      }
    });
  });

  // Countdown simulation
  setInterval(() => {
    if (countdownSeconds > 0) {
      countdownSeconds--;
      const mins = Math.floor(countdownSeconds / 60);
      const secs = countdownSeconds % 60;
      if (timerDisplay) {
        timerDisplay.textContent = `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
      }
    }
  }, 1000);

  // Confirm payment
  if (btnConfirmPayment) {
    btnConfirmPayment.addEventListener('click', () => {
      btnConfirmPayment.innerHTML = '⏳ កំពុងដំណើរការទូទាត់...';
      btnConfirmPayment.disabled = true;

      setTimeout(() => {
        btnConfirmPayment.innerHTML = '🔒 បញ្ជាក់ការទូទាត់ និងបញ្ចប់ការបញ្ជាទិញ';
        btnConfirmPayment.disabled = false;

        // Generate receipt values
        document.getElementById('order-id').textContent = `#ORD-2026-${Math.floor(1000 + Math.random() * 9000)}`;
        document.getElementById('receipt-date').textContent = new Date().toLocaleDateString('km-KH', { dateStyle: 'long' });
        document.getElementById('receipt-item').textContent = `TechPro Laptop (${appState.selectedColor}, ${appState.selectedStorage}) x ${appState.quantity}`;

        goToStep(5);
        showToast('🎉 ការទូទាត់ជោគជ័យ! បញ្ជាទិញបានសម្រេច។');
      }, 1200);
    });
  }
}

function initModalsAndHUD() {
  const toast = document.getElementById('toast-notification');
  const toastText = document.getElementById('toast-text');

  window.showToast = function(msg) {
    if (!toast || !toastText) return;
    toastText.textContent = msg;
    toast.classList.add('show');
    setTimeout(() => {
      toast.classList.remove('show');
    }, 3200);
  };

  // Flow Diagram Modal
  const flowModal = document.getElementById('flow-modal');
  const btnOpenFlow1 = document.getElementById('btn-view-userflow-diagram');
  const btnOpenFlow2 = document.getElementById('btn-show-flow-modal-2');
  const btnCloseFlow = document.getElementById('btn-close-flow-modal');
  const btnCloseFlowFooter = document.getElementById('btn-close-flow-footer');

  function openFlowModal() {
    if (flowModal) flowModal.classList.add('open');
  }

  function closeFlowModal() {
    if (flowModal) flowModal.classList.remove('open');
  }

  if (btnOpenFlow1) btnOpenFlow1.addEventListener('click', openFlowModal);
  if (btnOpenFlow2) btnOpenFlow2.addEventListener('click', openFlowModal);
  if (btnCloseFlow) btnCloseFlow.addEventListener('click', closeFlowModal);
  if (btnCloseFlowFooter) btnCloseFlowFooter.addEventListener('click', closeFlowModal);

  window.addEventListener('click', (e) => {
    if (e.target === flowModal) closeFlowModal();
  });

  // Toggle HUD minimize
  const btnToggleHud = document.getElementById('btn-toggle-hud');
  const hudContent = document.getElementById('hud-content');
  let hudMinimized = false;

  if (btnToggleHud && hudContent) {
    btnToggleHud.addEventListener('click', () => {
      hudMinimized = !hudMinimized;
      if (hudMinimized) {
        hudContent.style.display = 'none';
        btnToggleHud.textContent = '+';
      } else {
        hudContent.style.display = 'block';
        btnToggleHud.textContent = '_';
      }
    });
  }
}
