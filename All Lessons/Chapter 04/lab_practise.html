ជំពូកទី 04 • កិច្ចការអនុវត្តជាក់ស្តែង (Lab Practice)

១. ព័ត៌មានទូទៅ និងគោលបំណង (Lab Overview & Objectives)

កិច្ចការអនុវត្តបន្ទប់ពិសោធន៍នៃជំពូកទី 04 នេះ ត្រូវបានរៀបចំឡើងដើម្បីផ្ដល់ឱ្យនិស្សិតនូវជំនាញជាក់ស្ដែងក្នុងការសរសេរកូដ ការបង្កើតមគ្គុទ្ទេសក៍រចនាស្ទីលគេហទំព័រ (Create a Website Style Guide) ដោយអនុលោមតាមបទដ្ឋានឧស្សាហកម្មវេបទំនើប W3C និង UI/UX Best Practices។

🎯 គោលបំណងចម្បងនៃការអនុវត្ត៖
  • ស្វែងយល់ និងអនុវត្តជាក់ស្ដែងលើរចនាសម្ព័ន្ធកូដ៖ Color Theory, Typography, Visual Hierarchy, Font Family, Color Contrast
  • សរសេរកូដស្អាត (Clean Code) មានសណ្តាប់ធ្នាប់ និងឆ្លងកាត់ការផ្ទៀងផ្ទាត់ W3C Validator
  • ធានាភាពស៊ីសង្វាក់គ្នានៃការបង្ហាញលើ Mobile និង Desktop (Responsive Design)

function initContrastCalculator() {
  const textPicker = document.getElementById('color-text-picker');
  const textHex = document.getElementById('color-text-hex');
  const bgPicker = document.getElementById('color-bg-picker');
  const bgHex = document.getElementById('color-bg-hex');

  const previewBox = document.getElementById('contrast-preview-box');
  const ratioDisplay = document.getElementById('contrast-ratio-display');
  const auditNormal = document.getElementById('audit-normal');
  const auditLarge = document.getElementById('audit-large');

  // Relative luminance calculation according to W3C WCAG 2.1
  function getLuminance(hex) {
    const rgb = hexToRgb(hex);
    if (!rgb) return 0;
    const [r, g, b] = [rgb.r, rgb.g, rgb.b].map(val => {
      const v = val / 255;
      return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4);
    });
    return 0.2126 * r + 0.7152 * g + 0.0722 * b;
  }

  function hexToRgb(hex) {
    hex = hex.replace(/^#/, '');
    if (hex.length === 3) {
      hex = hex.split('').map(c => c + c).join('');
    }
    const num = parseInt(hex, 16);
    return isNaN(num) ? null : {
      r: (num >> 16) & 255,
      g: (num >> 8) & 255,
      b: num & 255
    };
  }

  function calculateContrast() {
    const tHex = textPicker.value;
    const bHex = bgPicker.value;

    // Update Hex textboxes
    textHex.value = tHex.toUpperCase();
    bgHex.value = bHex.toUpperCase();

    // Update preview box colors
    previewBox.style.color = tHex;
    previewBox.style.backgroundColor = bHex;

    // Compute contrast ratio
    const l1 = getLuminance(tHex);
    const l2 = getLuminance(bHex);
    const brighter = Math.max(l1, l2);
    const darker = Math.min(l1, l2);
    const ratio = (brighter + 0.05) / (darker + 0.05);
    const formattedRatio = ratio.toFixed(1);

    ratioDisplay.textContent = `${formattedRatio} : 1`;

    // Audit Normal Text (4.5:1 for AA)
    if (ratio >= 4.5) {
      auditNormal.innerHTML = `<span>Normal Text (4.5:1):</span> <strong class="pass-tag">✅ PASS (AA)</strong>`;
    } else {
      auditNormal.innerHTML = `<span>Normal Text (4.5:1):</span> <strong class="fail-tag">❌ FAIL</strong>`;
    }

    // Audit Large Text (3.0:1 for AA)
    if (ratio >= 3.0) {
      auditLarge.innerHTML = `<span>Large Text (3.0:1):</span> <strong class="pass-tag">✅ PASS (AA)</strong>`;
    } else {
      auditLarge.innerHTML = `<span>Large Text (3.0:1):</span> <strong class="fail-tag">❌ FAIL</strong>`;
    }
  }

  textPicker.addEventListener('input', calculateContrast);
  bgPicker.addEventListener('input', calculateContrast);

  textHex.addEventListener('change', () => {
    if (/^#[0-9A-F]{6}$/i.test(textHex.value)) {
      textPicker.value = textHex.value;
      calculateContrast();
    }
  });

  bgHex.addEventListener('change', () => {
    if (/^#[0-9A-F]{6}$/i.test(bgHex.value)) {
      bgPicker.value = bgHex.value;
      calculateContrast();
    }
  });

  // Initial calculation
  calculateContrast();
}

function initContrastCalculator() {
  const textPicker = document.getElementById('color-text-picker');
  const textHex = document.getElementById('color-text-hex');
  const bgPicker = document.getElementById('color-bg-picker');
  const bgHex = document.getElementById('color-bg-hex');

  const previewBox = document.getElementById('contrast-preview-box');
  const ratioDisplay = document.getElementById('contrast-ratio-display');
  const auditNormal = document.getElementById('audit-normal');
  const auditLarge = document.getElementById('audit-large');

  // Relative luminance calculation according to W3C WCAG 2.1
  function getLuminance(hex) {
    const rgb = hexToRgb(hex);
    if (!rgb) return 0;
    const [r, g, b] = [rgb.r, rgb.g, rgb.b].map(val => {
      const v = val / 255;
      return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4);
    });
    return 0.2126 * r + 0.7152 * g + 0.0722 * b;
  }

  function hexToRgb(hex) {
    hex = hex.replace(/^#/, '');
    if (hex.length === 3) {
      hex = hex.split('').map(c => c + c).join('');
    }
    const num = parseInt(hex, 16);
    return isNaN(num) ? null : {
      r: (num >> 16) & 255,
      g: (num >> 8) & 255,
      b: num & 255
    };
  }

  function calculateContrast() {
    const tHex = textPicker.value;
    const bHex = bgPicker.value;

    // Update Hex textboxes
    textHex.value = tHex.toUpperCase();
    bgHex.value = bHex.toUpperCase();

    // Update preview box colors
    previewBox.style.color = tHex;
    previewBox.style.backgroundColor = bHex;

    // Compute contrast ratio
    const l1 = getLuminance(tHex);
    const l2 = getLuminance(bHex);
    const brighter = Math.max(l1, l2);
    const darker = Math.min(l1, l2);
    const ratio = (brighter + 0.05) / (darker + 0.05);
    const formattedRatio = ratio.toFixed(1);

    ratioDisplay.textContent = `${formattedRatio} : 1`;

    // Audit Normal Text (4.5:1 for AA)
    if (ratio >= 4.5) {
      auditNormal.innerHTML = `<span>Normal Text (4.5:1):</span> <strong class="pass-tag">✅ PASS (AA)</strong>`;
    } else {
      auditNormal.innerHTML = `<span>Normal Text (4.5:1):</span> <strong class="fail-tag">❌ FAIL</strong>`;
    }

    // Audit Large Text (3.0:1 for AA)
    if (ratio >= 3.0) {
      auditLarge.innerHTML = `<span>Large Text (3.0:1):</span> <strong class="pass-tag">✅ PASS (AA)</strong>`;
    } else {
      auditLarge.innerHTML = `<span>Large Text (3.0:1):</span> <strong class="fail-tag">❌ FAIL</strong>`;
    }
  }

  textPicker.addEventListener('input', calculateContrast);
  bgPicker.addEventListener('input', calculateContrast);

  textHex.addEventListener('change', () => {
    if (/^#[0-9A-F]{6}$/i.test(textHex.value)) {
      textPicker.value = textHex.value;
      calculateContrast();
    }
  });

  bgHex.addEventListener('change', () => {
    if (/^#[0-9A-F]{6}$/i.test(bgHex.value)) {
      bgPicker.value = bgHex.value;
      calculateContrast();
    }
  });

  // Initial calculation
  calculateContrast();
}

៥. តារាងពិន្ទុវាយតម្លៃ ១០០ ពិន្ទុ (100-Point Evaluation Rubric)

លក្ខណៈវិនិច្ឆ័យ (Criteria) ទម្ងន់ពិន្ទុ សូចនាករវាស់វែងសមត្ថភាព (Performance Indicators)
១. រចនាសម្ព័ន្ធកូដ & HTML5 Semantics ២៥ ពិន្ទុ ប្រើប្រាស់ Semantic Elements ត្រឹមត្រូវតាមស្ដង់ដារ W3C, DOCTYPE, ឋានានុក្រម Headings (H1-H6) ច្បាស់លាស់ និងរៀបចំ Indentation ស្អាតបាត។
២. រចនាប័ទ្ម CSS & Responsive Design ២៥ ពិន្ទុ ប្រើប្រាស់ CSS Variables (:root), Box Model, Flexbox/Grid បានត្រឹមត្រូវ និងមាន Media Queries គាំទ្រការបង្ហាញលើ Mobile បានយ៉ាងរលូន។
៣. ភាពត្រឹមត្រូវនៃមុខងារអនុវត្ត ២៥ ពិន្ទុ សមាសភាគស្នូលនៃមេរៀន (ការបង្កើតមគ្គុទ្ទេសក៍រចនាស្ទីលគេហទំព័រ) ដំណើរការពេញលេញ គ្មាន Error និងឆ្លើយតបទៅនឹងគោលបំណងនៃមន្ទីរពិសោធន៍។
៤. សោភ័ណភាព UI/UX & ភាពងាយស្រួល ១៥ ពិន្ទុ ការជ្រើសរើសកម្រិតពណ៌ផ្ទុយត្រឹមត្រូវតាម WCAG, ការរៀបចំគម្លាត Spacing ស្មើដៃ និងទំហំអក្សរងាយស្រួលអានលើគ្រប់ឧបករណ៍។
